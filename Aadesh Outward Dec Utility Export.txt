[USER]
Doc md.txt is the existing utility which I built using plain Java, where the task
was to decrypt the files provided by NPCI containing XMLs. Now I want to build
another utility where the goal is to decrypt a combination of XMLs. After
unzipping, unsigning, and decryption, the result folder would have XMLs. Leave
the filename validation for now. Instead of a plain Java app, I want to use
Spring Boot and Java 21 for building this, following the organizational reference
structure. Start with giving me the project structure.

[SUMMARY]
Migrating the plain Java NPCI decryption utility to Spring Boot 3 and Java 21
shifts the architecture from manual thread management and raw JDBC to Spring
Data JPA, @Scheduled tasks, validated @ConfigurationProperties, and modular
services.

[USER]
The input will be a zip file and the output will be directly placed in the output
folder, no subfolder of zip name will be created in this one. The input files are
of 2 types:
  1. CREATE, CANCEL, AMEND files
  2. PAIN12 RES files
The output folder has 2 subfolders for these:
  1. Mandate
  2. Pain_12

[SUMMARY]
To route extracted XMLs directly into specific flat subfolders without creating
a ZIP-named parent folder:
- FilePathProperties binds explicit output.mandatePath (/Mandate) and
  output.pain12Path (/Pain_12) targets.
- FileRoutingService resolves the destination based on filename classification.
- ZipUtility performs flat extraction with UUID-based collision handling.

End-to-End Processing Flow
--------------------------------------------------------------------------------
[SFTP Server] --(SftpService)--> [Source Dir]
                                       |
                        (FilePollingScheduler @Scheduled)
                                       |
                                       v
                        [DecryptionOrchestratorService]
                         |-- 1. Duplicate check --> OUTWARD_DEC_LOGS
                         |-- 2. Route type --> FileRoutingService (Mandate vs Pain_12)
                         |-- 3. PGP Decrypt --> PgpCryptoUtility
                         |-- 4. Unsign/Verify --> SignatureVerificationService
                         |-- 5. Flat Unzip --> ZipUtility --> /Output/Mandate/ or /Output/Pain_12/
                         +-- 6. Audit & Backup --> OUTWARD_FILE_LOGS + OUTWARD_DEC_LOGS

package com.tcs.aadesh.scheduler;

import java.io.File;
import java.util.Arrays;
import java.util.concurrent.ExecutorService;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

import com.tcs.aadesh.config.properties.FilePathProperties;
import com.tcs.aadesh.service.DecryptionOrchestratorService;
import com.tcs.aadesh.service.SftpService;
import com.tcs.aadesh.util.ErrorLogger;

@Component
public class FilePollingScheduler {

	private static final Logger log = LoggerFactory.getLogger(FilePollingScheduler.class);

	private final FilePathProperties filePathProperties;
	private final SftpService sftpService;
	private final DecryptionOrchestratorService decryptionOrchestratorService;
	private final ExecutorService decryptionExecutor;

	@Autowired
	public FilePollingScheduler(FilePathProperties filePathProperties,
								SftpService sftpService,
								DecryptionOrchestratorService decryptionOrchestratorService,
								ExecutorService decryptionExecutor) {
		this.filePathProperties = filePathProperties;
		this.sftpService = sftpService;
		this.decryptionOrchestratorService = decryptionOrchestratorService;
		this.decryptionExecutor = decryptionExecutor;
	}

	@Scheduled(fixedDelayString = "${app.scheduler.poll-interval:60000}")
	public void pollAndProcessFiles() {
		try {
			// 1. Fetch files from remote SFTP to local source directory
			sftpService.fetchFilesFromRemote();

			// 2. Scan local source directory
			File sourceDir = new File(filePathProperties.getSourcePath());

			if (!sourceDir.exists() || !sourceDir.isDirectory()) {
				log.warn("Source directory does not exist or is invalid: {}", sourceDir.getAbsolutePath());
				return;
			}

			// Filter specifically for .zip extensions to avoid processing partial/temp files
			File[] files = sourceDir.listFiles((dir, name) -> name.toLowerCase().endsWith(".zip"));

			if (files == null || files.length == 0) {
				log.debug("No new ZIP files found in source directory.");
				return;
			}

			log.info("Found {} ZIP file(s) for processing.", files.length);

			// 3. Dispatch files to the Orchestrator concurrently via the ExecutorService
			Arrays.stream(files).forEach(file -> {
				decryptionExecutor.submit(() -> {
					try {
						decryptionOrchestratorService.processZipBatch(file);
					} catch (Exception e) {
						ErrorLogger.log(log, "Failed to process file: " + file.getName(), e);
					}
				});
			});

		} catch (Exception e) {
			ErrorLogger.log(log, "Error during scheduled file polling execution", e);
		}
	}
}

	@Autowired
	public DecryptionOrchestratorService(FilePathProperties filePathProperties,

										 FileRoutingService fileRoutingService,
										 SignatureVerificationService signatureVerificationService,
										 PgpCryptoUtility pgpCryptoUtility,
										 ZipUtility zipUtility,
										 OutwardDecLogRepository outwardDecLogRepository,
										 OutwardFileLogRepository outwardFileLogRepository) {
		this.filePathProperties = filePathProperties;
		this.cryptoProperties = cryptoProperties;
		this.fileRoutingService = fileRoutingService;
		this.signatureVerificationService = signatureVerificationService;
		this.pgpCryptoUtility = pgpCryptoUtility;
		this.zipUtility = zipUtility;

		this.outwardFileLogRepository = outwardFileLogRepository;

	public void processZipBatch(File sourceFile) {
		String fileName = sourceFile.getName();
		String referenceNumber = generateReferenceNumber();
		String zone = extractZone(fileName);
		Path tempDecryptedFile = null;

		log.info("Processing file: {} | Reference Number: {}", fileName, referenceNumber);

		OutwardDecLog decLog = new OutwardDecLog();
		decLog.setReferenceNumber(referenceNumber);
		decLog.setEntryDate(LocalDate.now());
		decLog.setFileName(fileName);
		decLog.setFileCopiedTime(LocalDateTime.now());
		decLog.setZone(zone);

		try {
			// 1. Duplicate check
			if (outwardDecLogRepository.existsByFileNameAndErrorCodeIsNull(fileName)) {
				log.warn("Duplicate file detected, moving to duplicate directory: {}", fileName);
				decLog.setStatus(ProcessingStatus.FAILED_FILE.getCode());
				decLog.setStatusDec(ProcessingStatus.FAILED_FILE.getDescription());
				decLog.setErrorCode(ProcessingStatus.ERR03.getCode());
				decLog.setLastUpdatedTime(LocalDateTime.now());
				outwardDecLogRepository.save(decLog);

				moveFileSafely(sourceFile.toPath(), filePathProperties.getFailed().getDuplicateFilePath());
				return;
			}

			// 2. Route file type & resolve flat output directory (Mandate vs Pain_12)
			FileType fileType = fileRoutingService.resolveFileType(fileName);
			Path targetOutputDir = fileRoutingService.resolveOutputDirectory(fileType);
			decLog.setFileType(fileType.getValue());

			// 3. Decrypt, unsign, and flat-unzip based on input format
			List<String> extractedXmlFiles;

			if (pgpCryptoUtility.isPgpEncrypted(sourceFile)) {
				decLog.setEncryptionType(ProcessingStatus.ENCRYPTED_FILE.getCode());
				decLog.setStatus(ProcessingStatus.FILE_PICKED_FOR_DECRYPTION.getCode());

				tempDecryptedFile = pgpCryptoUtility.decryptFile(sourceFile, filePathProperties.getDecryptedTempPath());
				decLog.setStatus(ProcessingStatus.DECRYPTED_UNSIGN_PENDING.getCode());
				decLog.setLastUpdatedTime(LocalDateTime.now());

				String base64OrgContent = signatureVerificationService.extractAndVerifySignature(tempDecryptedFile.toFile());
				decLog.setStatus(ProcessingStatus.UNSIGN_SUCCESS.getCode());
				decLog.setStatusDec(ProcessingStatus.UNSIGN_SUCCESS.getDescription());
				decLog.setLastUpdatedTime(LocalDateTime.now());

				byte[] zipBytes = Base64.getDecoder().decode(base64OrgContent);
				extractedXmlFiles = zipUtility.unzipFromBytesFlat(zipBytes, targetOutputDir);

			} else if (cryptoProperties.getPgp().isAllowUnencrypted()) {
				if (signatureVerificationService.isSignedFile(sourceFile)) {
					decLog.setEncryptionType(ProcessingStatus.SIGNED_FILE.getCode());
					String base64OrgContent = signatureVerificationService.extractAndVerifySignature(sourceFile);
					decLog.setStatus(ProcessingStatus.UNSIGN_SUCCESS.getCode());
					decLog.setStatusDec(ProcessingStatus.UNSIGN_SUCCESS.getDescription());
					decLog.setLastUpdatedTime(LocalDateTime.now());

					byte[] zipBytes = Base64.getDecoder().decode(base64OrgContent);
					extractedXmlFiles = zipUtility.unzipFromBytesFlat(zipBytes, targetOutputDir);

				} else if (zipUtility.isZipFile(sourceFile)) {
					decLog.setEncryptionType(ProcessingStatus.ZIPPED_FILE.getCode());
					extractedXmlFiles = zipUtility.unzipFileFlat(sourceFile, targetOutputDir);

				} else {
					throw new CryptoProcessingException(
							ProcessingStatus.DECRYPTION_FAILURE.getCode(),
							"File is neither PGP encrypted, signed XML, nor a valid ZIP: " + fileName);
				}
			} else {
				throw new CryptoProcessingException(
						ProcessingStatus.DECRYPTION_FAILURE.getCode(),
						"Unencrypted file rejected by configuration: " + fileName);
			}

			// 4. Log individual extracted XMLs to OUTWARD_FILE_LOGS
			String zipBaseName = stripExtension(fileName);
			saveExtractedFileLogs(extractedXmlFiles, zipBaseName, zone);

			// 5. Log batch summary to OUTWARD_DEC_LOGS
			decLog.setTotalRecords(extractedXmlFiles.size());
			decLog.setStatus(ProcessingStatus.DECRYPTION_SUCCESS.getCode());
			decLog.setStatusDec(ProcessingStatus.DECRYPTION_SUCCESS.getDescription());
			decLog.setLastUpdatedTime(LocalDateTime.now());
			outwardDecLogRepository.save(decLog);

			// 6. Backup source ZIP and extracted XMLs, then remove source ZIP
			backupFiles(sourceFile, extractedXmlFiles, targetOutputDir, fileType);
			Files.deleteIfExists(sourceFile.toPath());

			log.info("Successfully processed {} | Extracted {} XML(s) to {} | Ref: {}",
					fileName, extractedXmlFiles.size(), targetOutputDir, referenceNumber);

		} catch (UnrecognizedFileTypeException e) {
			ErrorLogger.log(log, "Unrecognized file type for: " + fileName, e);
			handleFailure(sourceFile, decLog, ProcessingStatus.FAILED_FILE, e.getErrorCode(),
					filePathProperties.getFailed().getUnrecognizedTypePath());

		} catch (CryptoProcessingException e) {
			ErrorLogger.log(log, "Crypto/extraction failure for: " + fileName, e);
			ProcessingStatus failedStatus = resolveStatusFromCode(e.getErrorCode());
			String failedDir = resolveFailedDirectory(e.getErrorCode());
			handleFailure(sourceFile, decLog, failedStatus, e.getErrorCode(), failedDir);

		} catch (Exception e) {
			ErrorLogger.log(log, "Unexpected error while processing: " + fileName, e);
			handleFailure(sourceFile, decLog, ProcessingStatus.DECRYPTION_FAILURE,
					ProcessingStatus.DECRYPTION_FAILURE.getCode(),
					filePathProperties.getFailed().getDecryptionFailedPath());

		} finally {
			if (tempDecryptedFile != null) {
				try {
					Files.deleteIfExists(tempDecryptedFile);
				} catch (IOException ex) {
					ErrorLogger.log(log, "Failed to delete temp decrypted file: " + tempDecryptedFile, ex);
				}
			}
		}
	}

	private void saveExtractedFileLogs(List<String> extractedXmlFiles, String zipBaseName, String zone) {
		LocalDate today = LocalDate.now();
		LocalDateTime now = LocalDateTime.now();

		List<OutwardFileLog> fileLogs = extractedXmlFiles.stream().map(xmlName -> {
			OutwardFileLog fileLog = new OutwardFileLog();
			fileLog.setXmlFileName(xmlName);
			fileLog.setFolderName(zipBaseName);
			fileLog.setZone(zone);
			fileLog.setEntryDate(today);
			fileLog.setLastUpdatedTime(now);
			fileLog.setOperationFlag("DEC");
			return fileLog;
		}).toList();

		outwardFileLogRepository.saveAll(fileLogs);
	}

	private void backupFiles(File sourceFile, List<String> extractedXmlFiles, Path targetOutputDir, FileType fileType)
			throws IOException {
		String dateFolder = LocalDate.now().format(BACKUP_DATE_FORMATTER);

		// Backup original ZIP
		Path zipBackupDir = Paths.get(filePathProperties.getBackupPath(), fileType.getValue(), dateFolder);
		Files.createDirectories(zipBackupDir);
		Files.copy(sourceFile.toPath(), zipBackupDir.resolve(sourceFile.getName()), StandardCopyOption.REPLACE_EXISTING);

		// Backup extracted XMLs
		Path xmlBackupDir = Paths.get(filePathProperties.getOutputBackupPath(), fileType.getValue(), dateFolder);
		Files.createDirectories(xmlBackupDir);
		for (String xmlFileName : extractedXmlFiles) {
			Path extractedFile = targetOutputDir.resolve(xmlFileName);
			if (Files.exists(extractedFile)) {
				Files.copy(extractedFile, xmlBackupDir.resolve(xmlFileName), StandardCopyOption.REPLACE_EXISTING);
			}
		}
	}

	private void handleFailure(File sourceFile, OutwardDecLog decLog, ProcessingStatus status, String errorCode,
			String failedDir) {
		decLog.setStatus(status.getCode());
		decLog.setStatusDec(status.getDescription());
		decLog.setErrorCode(errorCode);
		decLog.setLastUpdatedTime(LocalDateTime.now());
		outwardDecLogRepository.save(decLog);

		moveFileSafely(sourceFile.toPath(), failedDir);
	}

	private void moveFileSafely(Path source, String targetDir) {
		try {
			Path destinationDir = Paths.get(targetDir);
			Files.createDirectories(destinationDir);
			Files.move(source, destinationDir.resolve(source.getFileName()), StandardCopyOption.REPLACE_EXISTING);
		} catch (IOException ex) {
			ErrorLogger.log(log, "Failed to move file " + source.getFileName() + " to " + targetDir, ex);
		}
	}

	private ProcessingStatus resolveStatusFromCode(String code) {
		for (ProcessingStatus status : ProcessingStatus.values()) {
			if (status.getCode().equals(code)) {
				return status;
			}
		}
		return ProcessingStatus.DECRYPTION_FAILURE;
	}

	private String resolveFailedDirectory(String errorCode) {
		if (ProcessingStatus.UNSIGN_FAILURE.getCode().equals(errorCode)) {
			return filePathProperties.getFailed().getSignNotVerifiedPath();
		}
		if (ProcessingStatus.UNZIP_FAILURE.getCode().equals(errorCode)) {
			return filePathProperties.getFailed().getDecryptionFailedPath();
		}
		return filePathProperties.getFailed().getDecryptionFailedPath();
	}

	private String generateReferenceNumber() {
		String uuidDigits = String.format("%010d", new BigInteger(UUID.randomUUID().toString().replace("-", ""), 16));
		return "SBIN" + uuidDigits.substring(uuidDigits.length() - 10);
	}

	private String extractZone(String fileName) {
		if (fileName.contains("OTH")) {
			return "OTH";
		} else if (fileName.contains("SOU")) {
			return "SOU";
		} else if (fileName.contains("NOR")) {
			return "NOR";
		}
		return null;
	}

	private String stripExtension(String fileName) {
		int dotIndex = fileName.lastIndexOf('.');
		return dotIndex > 0 ? fileName.substring(0, dotIndex) : fileName;
	}
}

	public FileRoutingService(FilePathProperties filePathProperties) {
		this.filePathProperties = filePathProperties;
	}

	public FileType resolveFileType(String fileName) {
		if (fileName == null || fileName.isBlank()) {
			throw new UnrecognizedFileTypeException("Filename is null or empty");
		}

		String upperName = fileName.toUpperCase();

		if (upperName.contains("PAIN12")) {
			return FileType.PAIN12_RES;
		} else if (upperName.contains("CREATE")) {
			return FileType.CREATE;
		} else if (upperName.contains("AMEND")) {
			return FileType.AMEND;
		} else if (upperName.contains("CANCEL")) {
			return FileType.CANCEL;
		}

		log.warn("Could not determine file type for: {}", fileName);
		throw new UnrecognizedFileTypeException("Unrecognized file type for file: " + fileName);
	}

	public Path resolveOutputDirectory(FileType fileType) {
		if (fileType == null) {
			throw new UnrecognizedFileTypeException("FileType cannot be null");
		}

		Path targetDir = switch (fileType) {
			case CREATE, AMEND, CANCEL -> Paths.get(filePathProperties.getOutput().getMandatePath());
			case PAIN12_RES -> Paths.get(filePathProperties.getOutput().getPain12Path());
		};

		log.debug("Routed fileType {} to output directory: {}", fileType, targetDir);
		return targetDir;
	}

	public boolean isSignedFile(File file) {
		boolean has1 = false, has2 = false, has3 = false, has4 = false;
		try (BufferedReader reader = new BufferedReader(new FileReader(file))) {
			String line;
			while ((line = reader.readLine()) != null) {
				if (line.contains("<?xml version=\"1.0\" encoding=\"UTF-8\"") && line.contains("<Envelope><OrgContent>")) {
					has1 = true;
				}
				if (line.contains("</OrgContent><Signature>")) {
					has2 = true;
				}
				if (line.contains("</Signature><Certificate>")) {
					has3 = true;
				}
				if (line.contains("</Certificate></Envelope>")) {
					has4 = true;
				}
				if (has1 && has2 && has3 && has4) {
					return true;
				}
			}
		} catch (IOException e) {
			ErrorLogger.log(log, "Failed to read file for signature markers: " + file.getName(), e);
		}
		return false;
	}

	public String extractAndVerifySignature(File xmlFile) {
		try {
			// 1. Parse XML and extract Base64 content
			DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
			factory.setFeature("http://apache.org/xml/features/disallow-doctype-decl", true);
			DocumentBuilder builder = factory.newDocumentBuilder();
			Document doc = builder.parse(xmlFile);
			doc.getDocumentElement().normalize();

			Node orgContentNode = doc.getElementsByTagName("OrgContent").item(0);
			Node signatureNode = doc.getElementsByTagName("Signature").item(0);
			Node certificateNode = doc.getElementsByTagName("Certificate").item(0);

			if (orgContentNode == null || signatureNode == null || certificateNode == null) {
				throw new CryptoProcessingException(ProcessingStatus.UNSIGN_FAILURE.getCode(),
						"Missing required XML tags (OrgContent, Signature, Certificate) in file: " + xmlFile.getName());
			}

			String orgContentBase64 = orgContentNode.getTextContent();
			String signatureBase64 = signatureNode.getTextContent();
			String certificateBase64 = certificateNode.getTextContent();

			// 2. Decode and save the certificate for audit
			byte[] certBytes = Base64.getDecoder().decode(certificateBase64);
			Path certOutputPath = Paths.get(cryptoProperties.getPgp().getCheckingCertPath());
			Files.write(certOutputPath, certBytes);

			// 3. Generate X509Certificate object
			X509Certificate x509Cert;
			try (InputStream certStream = new ByteArrayInputStream(certBytes)) {
				CertificateFactory certFactory = CertificateFactory.getInstance("X.509");
				x509Cert = (X509Certificate) certFactory.generateCertificate(certStream);
			}

			// 4. Verify CMS Signature
			byte[] originalContentBytes = Base64.getDecoder().decode(orgContentBase64);
			byte[] signatureBytes = Base64.getDecoder().decode(signatureBase64);

			CMSProcessableByteArray cmsContent = new CMSProcessableByteArray(originalContentBytes);
			CMSSignedData signedData = new CMSSignedData(cmsContent, signatureBytes);
			SignerInformationStore signers = signedData.getSignerInfos();
			Collection<SignerInformation> signerInfos = signers.getSigners();

			boolean isSignValid = false;
			for (SignerInformation signer : signerInfos) {
				if (signer.verify(new JcaSimpleSignerInfoVerifierBuilder().build(x509Cert))) {
					isSignValid = true;
					break;
				}
			}

			if (!isSignValid) {
				throw new CryptoProcessingException(ProcessingStatus.UNSIGN_FAILURE.getCode(),
						"CMS Signature verification failed against the embedded certificate for file: " + xmlFile.getName());
			}

			return orgContentBase64;

		} catch (Exception e) {
			ErrorLogger.log(log, "Signature extraction/verification error for file: " + xmlFile.getName(), e);
			throw new CryptoProcessingException(ProcessingStatus.UNSIGN_FAILURE.getCode(),
					"Error during signature verification: " + e.getMessage(), e);
		}
	}
}

import java.io.File;

import org.slf4j.Logger;

import com.jcraft.jsch.Session;
import com.jcraft.jsch.SftpATTRS;
import com.tcs.aadesh.config.properties.SftpProperties;
import com.tcs.aadesh.exception.SftpTransferException;
import com.tcs.aadesh.util.AesCredentialUtility;
import com.tcs.aadesh.util.ErrorLogger;

@Service
public class SftpService {

	private static final Logger log = LoggerFactory.getLogger(SftpService.class);

	private final SftpProperties sftpProperties;
	private final AesCredentialUtility aesCredentialUtility;

	@Autowired
	public SftpService(SftpProperties sftpProperties, AesCredentialUtility aesCredentialUtility) {
		this.sftpProperties = sftpProperties;
		this.aesCredentialUtility = aesCredentialUtility;
	}

	public void fetchFilesFromRemote() {
		if (!sftpProperties.isEnabled()) {
			log.debug("SFTP transfer is disabled in properties.");
			return;
		}

		Session session = null;
		ChannelSftp channel = null;

		try {
			String username = aesCredentialUtility.decrypt(sftpProperties.getUsername());
			String privateKeyPath = aesCredentialUtility.decrypt(sftpProperties.getPrivateKeyPath());

			JSch jsch = new JSch();
			jsch.addIdentity(privateKeyPath.trim());

			session = jsch.getSession(username, sftpProperties.getHost(), sftpProperties.getPort());

			java.util.Properties config = new java.util.Properties();
			config.put("StrictHostKeyChecking", "no");
			config.put("PreferredAuthentications", "publickey");
			session.setConfig(config);

			log.info("Connecting to SFTP server {} on port {}...", sftpProperties.getHost(), sftpProperties.getPort());
			session.connect();

			channel = (ChannelSftp) session.openChannel("sftp");
			channel.connect();

			log.info("SFTP connection established successfully.");

			int filesDownloaded = downloadAndArchiveFiles(channel);
			log.info("SFTP fetch complete. Total files downloaded: {}", filesDownloaded);

		} catch (Exception e) {
			ErrorLogger.log(log, "Failed to fetch files from remote SFTP server", e);
			throw new SftpTransferException("Error during SFTP operations: " + e.getMessage(), e);
		} finally {
			disconnect(channel, session);
		}
	}

	private int downloadAndArchiveFiles(ChannelSftp channel) throws Exception {
		String remoteDir = sftpProperties.getRemoteDir();
		String localDir = sftpProperties.getLocalDir();

		channel.cd(remoteDir);

		@SuppressWarnings("unchecked")
		Vector<LsEntry> fileList = channel.ls("*.zip");

		int count = 0;

		if (fileList != null && !fileList.isEmpty()) {
			for (LsEntry entry : fileList) {
				if (!entry.getAttrs().isDir()) {
					String filename = entry.getFilename();
					String remoteFilePath = remoteDir + (remoteDir.endsWith("/") ? "" : "/") + filename;
					String localFilePath = localDir + File.separator + filename;

					log.info("Downloading file: {} to {}", remoteFilePath, localFilePath);

					channel.get(remoteFilePath, localFilePath);
					moveRemoteFileToArchive(channel, remoteFilePath, filename);
					count++;
				}
			}
		} else {
			log.debug("No .zip files found in remote directory: {}", remoteDir);
		}

		return count;
	}

	private void moveRemoteFileToArchive(ChannelSftp channel, String sourcePath, String filename) {
		try {
			String archiveDir = sftpProperties.getArchiveDir();
			String archivePath = archiveDir + (archiveDir.endsWith("/") ? "" : "/") + filename;

			SftpATTRS attrs = channel.stat(sourcePath);
			if (attrs != null) {
				channel.rename(sourcePath, archivePath);
				log.info("Moved remote file {} to archive {}", sourcePath, archivePath);
			} else {
				log.error("Source file does not exist on remote for archiving: {}", sourcePath);
			}
		} catch (Exception e) {
			ErrorLogger.log(log, "Failed to archive remote file: " + sourcePath, e);
		}
	}

	private void disconnect(ChannelSftp channel, Session session) {
		if (channel != null && channel.isConnected()) {
			channel.disconnect();
		}
		if (session != null && session.isConnected()) {
			session.disconnect();
		}
	}
}

package com.tcs.aadesh.util;

import java.nio.charset.StandardCharsets;
import java.util.Base64;

import javax.crypto.Cipher;
import javax.crypto.spec.GCMParameterSpec;
import javax.crypto.spec.SecretKeySpec;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import com.tcs.aadesh.config.properties.CryptoProperties;
import com.tcs.aadesh.exception.CryptoProcessingException;

@Component
public class AesCredentialUtility {

	private static final Logger log = LoggerFactory.getLogger(AesCredentialUtility.class);
	private static final String AES_TRANSFORMATION = "AES/GCM/NoPadding";

	private final CryptoProperties cryptoProperties;

	@Autowired
	public AesCredentialUtility(CryptoProperties cryptoProperties) {
		this.cryptoProperties = cryptoProperties;
	}

	public String decrypt(String encrypted) {
		if (encrypted == null || encrypted.isBlank()) {
			return encrypted;
		}

		try {
			byte[] keyBytes = Base64.getDecoder().decode(cryptoProperties.getAes().getKey().getBytes(StandardCharsets.UTF_8));
			byte[] ivBytes = Base64.getDecoder().decode(cryptoProperties.getAes().getIv().getBytes(StandardCharsets.UTF_8));

			SecretKeySpec secretKeySpec = new SecretKeySpec(keyBytes, "AES");
			GCMParameterSpec gcmParameterSpec = new GCMParameterSpec(128, ivBytes);

			Cipher cipher = Cipher.getInstance(AES_TRANSFORMATION);
			cipher.init(Cipher.DECRYPT_MODE, secretKeySpec, gcmParameterSpec);

			byte[] decodedEncryptedBytes = Base64.getDecoder().decode(encrypted);
			byte[] originalBytes = cipher.doFinal(decodedEncryptedBytes);

			return new String(originalBytes, StandardCharsets.UTF_8).replace("\0", "");

		} catch (Exception e) {
			ErrorLogger.log(log, "Failed to decrypt AES credential", e);
			throw new CryptoProcessingException("AES_DECRYPTION_ERROR", "Failed to decrypt credential property", e);
		}
	}
}

package com.tcs.aadesh.util;

import org.slf4j.Logger;
import org.springframework.stereotype.Component;

@Component
public final class ErrorLogger {

	static String currentPackage = ErrorLogger.class.getPackage().getName();
	static int lastDotIndex = currentPackage.lastIndexOf('.');
	private static final String APP_BASE_PACKAGE = currentPackage.substring(0, lastDotIndex);

	private ErrorLogger() {
	}

	public static void log(Logger log, Exception e) {
		if (e == null) {
			log.error("Unknown error (exception was null)");
			return;
		}

		StackTraceElement origin = findApplicationTrace(e);

		log.error("{} in {}.{}({}:{}) | {}",
				e.getClass().getSimpleName(),
				origin.getClassName(),
				origin.getMethodName(),
				origin.getFileName(),
				origin.getLineNumber(),
				safeMessage(e));
	}

	public static void log(Logger log, String message, Exception e) {
		if (e == null) {
			log.error("{} | Unknown error (exception was null)", message);
			return;
		}

		StackTraceElement origin = findApplicationTrace(e);

		log.error("{} | {} in {}.{}({}:{}) | {}",
				message,
				e.getClass().getSimpleName(),
				origin.getClassName(),
				origin.getMethodName(),
				origin.getFileName(),
				origin.getLineNumber(),
				safeMessage(e));
	}

	private static StackTraceElement findApplicationTrace(Exception e) {
		for (StackTraceElement element : e.getStackTrace()) {
			if (element.getClassName().startsWith(APP_BASE_PACKAGE)) {
				return element;
			}
		}
		return e.getStackTrace().length > 0 ? e.getStackTrace()[0]
				: new StackTraceElement("UnknownClass", "UnknownMethod", "UnknownFile", -1);
	}

	private static String safeMessage(Exception e) {
		return (e.getMessage() == null || e.getMessage().isBlank()) ? "(no message)" : e.getMessage();
	}
}

import com.tcs.aadesh.config.properties.CryptoProperties;

import com.tcs.aadesh.exception.CryptoProcessingException;

import jakarta.annotation.PostConstruct;

@Component
public class PgpCryptoUtility {

	private static final Logger log = LoggerFactory.getLogger(PgpCryptoUtility.class);

	private final CryptoProperties cryptoProperties;
	private PGPPrivateKey pgpPrivateKey;

	@Autowired
	public PgpCryptoUtility(CryptoProperties cryptoProperties) {
		this.cryptoProperties = cryptoProperties;
	}

	@PostConstruct
	public void init() {
		try {
			Security.addProvider(new BouncyCastleProvider());

			// 1. Load Public Key (Certificate)
			CertificateFactory certFactory = CertificateFactory.getInstance("X.509");
			PublicKey pubKey;
			try (FileInputStream fis = new FileInputStream(cryptoProperties.getPgp().getPublicKeyPath())) {
				Certificate certificate = certFactory.generateCertificate(fis);
				pubKey = certificate.getPublicKey();
			}

			PGPPublicKey pgpPubKey = new JcaPGPKeyConverter()
					.getPGPPublicKey(PGPPublicKey.RSA_GENERAL, pubKey, new java.util.Date());

			// 2. Load Private Key (PKCS12)
			KeyStore keyStore = KeyStore.getInstance("PKCS12");
			char[] password = cryptoProperties.getPgp().getPassword().toCharArray();
			try (FileInputStream fis = new FileInputStream(cryptoProperties.getPgp().getPrivateKeyPath())) {
				keyStore.load(fis, password);
			}

			String alias = keyStore.aliases().nextElement();
			PrivateKey privKey = (PrivateKey) keyStore.getKey(alias, password);

			// 3. Generate PGPPrivateKey
			this.pgpPrivateKey = new JcaPGPKeyConverter().getPGPPrivateKey(pgpPubKey, privKey);

			log.info("PGP Keys loaded and BouncyCastle initialized successfully.");

		} catch (Exception e) {
			ErrorLogger.log(log, "Failed to initialize PGP Keys", e);
			throw new IllegalStateException("Could not initialize PGP Cryptography components", e);
		}
	}

	public boolean isPgpEncrypted(File file) {
		boolean hasBeginMarker = false;
		boolean hasEndMarker = false;

		try (BufferedReader reader = new BufferedReader(new FileReader(file))) {
			String line;
			while ((line = reader.readLine()) != null) {
				if (line.contains("-----BEGIN PGP MESSAGE-----")) {
					hasBeginMarker = true;
				}
				if (line.contains("-----END PGP MESSAGE-----")) {
					hasEndMarker = true;
				}
				if (hasBeginMarker && hasEndMarker) {
					return true;
				}
			}
		} catch (IOException e) {
			ErrorLogger.log(log, "Failed to read file while checking for PGP markers: " + file.getName(), e);
		}
		return false;
	}

	public Path decryptFile(File encryptedFile, String targetDecryptedDir) {
		Path targetPath = Paths.get(targetDecryptedDir, encryptedFile.getName().replaceAll("\\.zip$", ".txt"));

		try (InputStream in = new BufferedInputStream(new FileInputStream(encryptedFile))) {

			InputStream decoderStream = PGPUtil.getDecoderStream(in);
			PGPObjectFactory pgpFactory = new PGPObjectFactory(decoderStream, new JcaKeyFingerprintCalculator());

			Object object = pgpFactory.nextObject();
			PGPEncryptedDataList encryptedDataList = (object instanceof PGPEncryptedDataList)
					? (PGPEncryptedDataList) object
					: (PGPEncryptedDataList) pgpFactory.nextObject();

			if (encryptedDataList == null) {
				throw new PGPException("Could not find PGPEncryptedDataList in the file.");
			}

			Iterator<PGPEncryptedData> it = encryptedDataList.getEncryptedDataObjects();
			PGPPublicKeyEncryptedData pbe = null;
			PGPPrivateKey sKey = null;

			while (it.hasNext() && sKey == null) {
				pbe = (PGPPublicKeyEncryptedData) it.next();
				sKey = this.pgpPrivateKey;
			}

			if (sKey == null || pbe == null) {
				throw new IllegalArgumentException("Secret key for message not found.");
			}

			PublicKeyDataDecryptorFactory decryptorFactory = new JcePublicKeyDataDecryptorFactoryBuilder()
					.setProvider("BC")
					.setContentProvider("BC")
					.build(sKey);

			InputStream clearStream = pbe.getDataStream(decryptorFactory);
			if (clearStream == null) {
				throw new IOException("Failed to obtain data stream from PGPPublicKeyEncryptedData.");
			}

			PGPObjectFactory plainFact = new PGPObjectFactory(clearStream, new JcaKeyFingerprintCalculator());
			Object message = plainFact.nextObject();

			if (message instanceof PGPCompressedData) {
				PGPCompressedData cData = (PGPCompressedData) message;
				PGPObjectFactory pgpFact = new PGPObjectFactory(cData.getDataStream(), new JcaKeyFingerprintCalculator());
				message = pgpFact.nextObject();
			}

			if (message instanceof CMSSignedData) {
				CMSSignedData cms = (CMSSignedData) message;
				byte[] contentBytes = (byte[]) cms.getSignedContent().getContent();
				FileUtils.writeByteArrayToFile(targetPath.toFile(), contentBytes);

			} else if (message instanceof PGPLiteralData) {
				PGPLiteralData ld = (PGPLiteralData) message;
				try (InputStream uncStream = ld.getInputStream()) {
					FileUtils.copyInputStreamToFile(uncStream, targetPath.toFile());
				}

			} else if (message instanceof PGPOnePassSignatureList) {
				throw new PGPException("Encrypted message contains a signed message - not literal data.");
			} else {
				throw new PGPException("Message is not a simple encrypted file - type unknown.");
			}

			if (pbe.isIntegrityProtected() && !pbe.verify()) {
				throw new PGPException("Message failed integrity check. Data may be corrupted or tampered with.");
			}

			return targetPath;

		} catch (PGPException | IOException e) {
			ErrorLogger.log(log, "PGP Decryption failed for file: " + encryptedFile.getName(), e);
			throw new CryptoProcessingException(ProcessingStatus.DECRYPTION_FAILURE.getCode(),
					"Decryption failed: " + e.getMessage(), e);
		}
	}
}

package com.tcs.aadesh.util;

import java.io.ByteArrayInputStream;
import java.io.File;
import java.io.IOException;
import java.io.InputStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.nio.file.StandardCopyOption;
import java.util.ArrayList;
import java.util.Enumeration;
import java.util.List;
import java.util.UUID;
import java.util.zip.ZipEntry;
import java.util.zip.ZipException;
import java.util.zip.ZipFile;
import java.util.zip.ZipInputStream;

import org.apache.commons.io.FilenameUtils;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;

import com.tcs.aadesh.constant.ProcessingStatus;
import com.tcs.aadesh.exception.CryptoProcessingException;

@Component
public class ZipUtility {

	private static final Logger log = LoggerFactory.getLogger(ZipUtility.class);

	public boolean isZipFile(File file) {
		try (ZipFile zipFile = new ZipFile(file)) {
			return true;
		} catch (ZipException e) {
			log.debug("The given file is not a valid zip file: {}", file.getName());
			return false;
		} catch (IOException e) {
			ErrorLogger.log(log, "I/O Error while checking if file is zip: " + file.getName(), e);
			return false;
		}
	}

	public List<String> unzipFromBytesFlat(byte[] zipBytes, Path targetOutputDir) {
		List<String> extractedXmlFiles = new ArrayList<>();

		try {
			Files.createDirectories(targetOutputDir);
		} catch (IOException e) {
			throw new CryptoProcessingException(ProcessingStatus.UNZIP_FAILURE.getCode(),
					"Failed to create target output directory: " + targetOutputDir, e);
		}

		try (InputStream byteArrayInputStream = new ByteArrayInputStream(zipBytes);
			 ZipInputStream zis = new ZipInputStream(byteArrayInputStream)) {

			ZipEntry zipEntry;
			while ((zipEntry = zis.getNextEntry()) != null) {
				if (!zipEntry.isDirectory()) {
					String extractedName = extractAndWriteEntry(zis, zipEntry.getName(), targetOutputDir);
					extractedXmlFiles.add(extractedName);
				}
				zis.closeEntry();
			}

		} catch (IOException e) {
			ErrorLogger.log(log, "Failed to unzip from byte array.", e);
			throw new CryptoProcessingException(ProcessingStatus.UNZIP_FAILURE.getCode(),
					"Failed to extract ZIP contents from memory.", e);
		}

		return extractedXmlFiles;
	}

	public List<String> unzipFileFlat(File zipFile, Path targetOutputDir) {
		List<String> extractedXmlFiles = new ArrayList<>();

		try {
			Files.createDirectories(targetOutputDir);
		} catch (IOException e) {
			throw new CryptoProcessingException(ProcessingStatus.UNZIP_FAILURE.getCode(),
					"Failed to create target output directory: " + targetOutputDir, e);
		}

		try (ZipFile zf = new ZipFile(zipFile)) {
			Enumeration<? extends ZipEntry> entries = zf.entries();

			while (entries.hasMoreElements()) {
				ZipEntry entry = entries.nextElement();
				if (!entry.isDirectory()) {
					try (InputStream is = zf.getInputStream(entry)) {
						String extractedName = extractAndWriteEntry(is, entry.getName(), targetOutputDir);
						extractedXmlFiles.add(extractedName);
					}
				}
			}

		} catch (IOException e) {
			ErrorLogger.log(log, "Failed to unzip physical file: " + zipFile.getName(), e);
			throw new CryptoProcessingException(ProcessingStatus.UNZIP_FAILURE.getCode(),
					"Failed to extract physical ZIP file: " + zipFile.getName(), e);
		}

		return extractedXmlFiles;
	}

	private String extractAndWriteEntry(InputStream inputStream, String entryName, Path targetOutputDir)
			throws IOException {
		// Flatten the path: extract only the filename, ignoring any folders inside the zip
		String fileName = Paths.get(entryName).getFileName().toString();
		Path targetFilePath = targetOutputDir.resolve(fileName);

		// Collision Handling: append a short UUID if a file with the same name already exists
		if (Files.exists(targetFilePath)) {
			String baseName = FilenameUtils.getBaseName(fileName);
			String extension = FilenameUtils.getExtension(fileName);
			String uniqueSuffix = UUID.randomUUID().toString().substring(0, 6);
			fileName = extension.isEmpty()
					? baseName + "_" + uniqueSuffix
					: baseName + "_" + uniqueSuffix + "." + extension;
			targetFilePath = targetOutputDir.resolve(fileName);
			log.info("Filename collision detected. Renamed extracted file to: {}", fileName);
		}

		Files.copy(inputStream, targetFilePath, StandardCopyOption.REPLACE_EXISTING);
		return fileName;
	}
}
